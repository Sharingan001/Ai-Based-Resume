cmake_minimum_required(VERSION 3.16)
project(nexus_score LANGUAGES CXX)
add_executable(nexus_score main.cpp)
target_compile_features(nexus_score PRIVATE cxx_std_17)
enable_testing()
add_test(NAME weighted_score COMMAND nexus_score 80 100 50 75)
set_tests_properties(weighted_score PROPERTIES PASS_REGULAR_EXPRESSION "81.5")
add_test(NAME rejects_negative COMMAND nexus_score -1 100 50 75)
set_tests_properties(rejects_negative PROPERTIES WILL_FAIL TRUE)
